def load_layer_data(atlas='BN', level='subclass', return_ratio=True):
    """Load per-layer cell-type counts and return a canonical stack.

    Returns
    -------
    arr : np.ndarray, shape (n_region, n_layer, n_ctype)
    ctype_names : list[str]
    region_names : list[str]
    layers : list[str]
    """
    path = ('/data100/home/dthbca/project/CellAlign/CellAlign/datasets/'
            'features/SpatialTranscriptomics/layer_data')
    data_list, region_names, ctype_names = [], None, None

    for layer in LAYER_KEYS:
        fname = f'ctype_region_counts_D99_M1_{layer}.csv'
        data = pd.read_csv(os.path.join(path, fname), index_col=0)
        if atlas != 'D99':
            data = vol_relabel('D99', atlas, data, method='sum',
                               cross_species=True).round()
        data = ctype_ratio_agg(data, map_df=ctype_map, key=level)
        if return_ratio:
            data = data.div(data.sum(axis=1), axis=0)
        data = data.sort_index().fillna(0)
        region_names = set(data.index) if region_names is None \
            else region_names.union(set(data.index))
        if ctype_names is None:
            ctype_names = data.columns.tolist()
        elif not all(data.columns == ctype_names):
            data = data[ctype_names]
        data_list.append(data)

    region_names = sorted(region_names)
    arr = np.stack([d.reindex(index=region_names, fill_value=0).values
                    for d in data_list], axis=0)      # (layer, region, ctype)
    arr = np.transpose(arr, (1, 0, 2))                # (region, layer, ctype)  <- canonical
    return arr, ctype_names, region_names, LAYER_KEYS

from scipy.stats import gmean

def coda_transform(df, mode='ALR', ref=None, pseudocount=1e-8, basis=None):
    """ALR / CLR / ILR transform for compositional data (rows = samples).

    CLR is used for the primary analysis. ILR uses the default SBP (Helmert-type)
    orthonormal basis. Returns (transformed_df, reference_or_basis).
    """
    cols = df.columns.tolist()
    if len(cols) < 2:
        raise ValueError('Need at least two components.')
    data = df[cols] + pseudocount
    D = len(cols)

    if mode.upper() == 'ALR':
        ref_col = (data.median(axis=0).idxmax() if ref is None
                   else cols[ref] if isinstance(ref, int) else ref)
        if ref_col not in cols:
            raise ValueError(f"Reference column '{ref_col}' does not exist.")
        out = pd.DataFrame({c: np.log(data[c] / data[ref_col])
                            for c in cols if c != ref_col}, index=df.index)
        return out, ref_col

    if mode.upper() == 'CLR':
        g = gmean(data, axis=1)
        out = pd.DataFrame({c: np.log(data[c] / g) for c in cols}, index=df.index)
        return out, 'Geometric mean'

    if mode.upper() == 'ILR':
        if basis is not None:
            B = np.asarray(basis)
            if B.shape != (D, D - 1):
                raise ValueError(f'Basis must be ({D}, {D-1}), got {B.shape}')
        else:
            B = np.zeros((D, D - 1))
            for i in range(1, D):
                coef = np.sqrt(i / (i + 1))
                B[:i, i - 1] = coef / i
                B[i,  i - 1] = -coef
        g = gmean(data, axis=1)
        clr = np.log(data / g.reshape(-1, 1))
        out = pd.DataFrame(clr.values @ B, index=df.index,
                           columns=[f'ILR_{i+1}' for i in range(D - 1)])
        return out, B

    raise ValueError("mode must be 'ALR', 'CLR', or 'ILR'")


def clr_features(prop_mat, layers, ctypes, mask=None, use_clr=True, pseudocount=1e-8):
    """Per-layer feature DataFrames used by the coupling tests.

    If mask (layers x ctypes bool, True=present) is given AND use_clr, each layer's
    CLR is computed over ONLY its present cell types (structural zeros excluded so
    they do not pollute the geometric mean). Returns list[n_layer] of (region x k_valid).
    """
    X = []
    for l, layer in enumerate(layers):
        df = pd.DataFrame(prop_mat[:, l, :], columns=ctypes)
        if mask is not None:
            keep = mask.loc[layer]
            df = df.loc[:, keep[keep].index]          # subcomposition: drop absent cols
        X.append(coda_transform(df, mode='CLR', pseudocount=pseudocount)[0]
                 if use_clr else df)
    return X

def plot_layer_heatmap(data, data_p=None, annot=False,
                       annot_thresholds=(0.05, 0.01, 0.001),
                       annot_symbols=('*', '**', '***'),
                       figsize_scale=(0.31, 0.4), vmin=-2, vmax=2,
                       cmap_name='RdYlBu_r', mask_color='#E8E8E8',
                       cbar_label='Layer enrichment score',
                       title='Cortical layer cell type distribution',
                       add_grid=True, output_file=None, dpi=300, mask=None):
    """Nature-style layer x cell-type heatmap.

    data : DataFrame (rows = layers, cols = cell types). NaN cells shown as mask_color.
    mask : optional bool DataFrame; True cells are hidden (set to NaN).
    """
    # 1. column ordering by class ------------------------------------------------
    cls_map = {ct: cls for cls in cls_order for ct in cls_def[cls]}
    base_to_cols = {}
    for col in data.columns:
        base = col if col in cls_map else (col.rsplit('_', 1)[0] if '_' in col else col)
        base_to_cols.setdefault(base, []).append(col)
    final_cols = []
    for cls in cls_order:
        for ct in cls_def[cls]:
            if ct in base_to_cols:
                final_cols.extend(base_to_cols.pop(ct))
    other = [c for cols in base_to_cols.values() for c in cols]
    if other:
        print(f'Warning: unclassified columns placed in Other: {other}')
        final_cols.extend(other)
    final_cols = [c for c in final_cols if c in data.columns]

    data = data[final_cols].copy(); data.index = layer_names_roman
    if data_p is not None:
        data_p = data_p[final_cols].copy(); data_p.index = layer_names_roman

    # 1b. apply mask -------------------------------------------------------------
    if mask is not None:
        if isinstance(mask, np.ndarray):
            mask = pd.DataFrame(mask, columns=final_cols)
        mask = mask[final_cols].copy(); mask.index = layer_names_roman
        data = data.mask(mask.values)
        if data_p is not None:
            data_p = data_p.mask(mask.values)

    n_rows, n_cols = data.shape

    # 2. class -> column spans for top bar --------------------------------------
    class_columns = OrderedDict()
    for col in final_cols:
        base = col if col in cls_map else (col.rsplit('_', 1)[0] if '_' in col else col)
        class_columns.setdefault(cls_map.get(base, 'Other'), []).append(col)
    cls_positions, start = {}, 0
    for cls, cols in class_columns.items():
        cls_positions[cls] = (start, start + len(cols)); start += len(cols)

    # 3-4. norm / figure ---------------------------------------------------------
    norm = TwoSlopeNorm(vmin=vmin, vcenter=(vmin + vmax) / 2, vmax=vmax)
    cmap = plt.get_cmap(cmap_name).copy(); cmap.set_bad(color=mask_color)
    fig, ax = plt.subplots(figsize=(n_cols * figsize_scale[0] + 1.2,
                                    n_rows * figsize_scale[1] + 0.8))
    im = ax.imshow(data.values, cmap=cmap, norm=norm, aspect='auto',
                   interpolation='nearest')
    if add_grid:
        for i in range(1, n_rows): ax.axhline(i - 0.52, color='white', lw=0.6, zorder=10)
        for j in range(1, n_cols): ax.axvline(j - 0.52, color='white', lw=0.6, zorder=10)

    # 5b. significance stars -----------------------------------------------------
    if annot and data_p is not None:
        for i in range(n_rows):
            for j in range(n_cols):
                val, p = data.iloc[i, j], data_p.iloc[i, j]
                if np.isnan(val) or np.isnan(p):
                    continue
                sym = next((s for t, s in zip(annot_thresholds, annot_symbols) if p < t), None)
                if sym is None:
                    continue
                rgba = cmap(norm(val))
                lum = 0.299 * rgba[0] + 0.587 * rgba[1] + 0.114 * rgba[2]
                ax.text(j, i, sym, ha='center', va='center', zorder=20,
                        color='white' if lum < 0.5 else 'black',
                        fontsize=8, fontweight='bold')

    # 6. axes --------------------------------------------------------------------
    ax.set_xticks(np.arange(n_cols)); ax.set_xticklabels(final_cols, rotation=90, fontsize=8)
    ax.yaxis.tick_right(); ax.yaxis.set_label_position('right')
    ax.set_yticks(np.arange(n_rows)); ax.set_yticklabels(layer_names_roman, fontsize=8, va='center')
    ax.tick_params(axis='y', length=0)
    for sp in ax.spines.values(): sp.set_visible(False)

    # 7. top class bar -----------------------------------------------------------
    top = ax.inset_axes([0, 1.02, 1, 0.1], transform=ax.transAxes)
    top.set_xlim(0, n_cols); top.set_ylim(0, 1); top.axis('off')
    for cls, (s, e) in cls_positions.items():
        top.add_patch(mpatches.Rectangle((s, 0), e - s, 1,
                      facecolor=cls_colors.get(cls, '#888'), edgecolor='none'))
        if e - s > 2:
            top.text((s + e) / 2, 0.5, cls, ha='center', va='center',
                     fontsize=9, color='white', fontweight='bold')

    # 8-9. colorbar / title ------------------------------------------------------
    cbar = fig.colorbar(im, ax=ax, orientation='vertical', fraction=0.1, pad=0.08,
                        shrink=0.8, aspect=10)
    cbar.set_label(cbar_label, fontsize=9, labelpad=8)
    cbar.ax.tick_params(labelsize=7, length=3, width=0.6)
    ticks = np.arange(vmin, vmax + 0.01, (vmax - vmin) / 4)
    cbar.set_ticks(ticks); cbar.ax.set_yticklabels([f'{t:.2f}' for t in ticks])
    ax.set_title(title, fontsize=11, pad=18, weight='normal')

    plt.tight_layout()
    if output_file:
        plt.savefig(output_file, dpi=dpi, bbox_inches='tight')
    plt.show()
    return fig, ax

from neuromaps import stats
from statsmodels.stats.multitest import fdrcorrection
from concurrent.futures import ProcessPoolExecutor, as_completed

def _spin_single(args):
    i, k, layer, ctype, x, y, spins = args
    r, p = stats.compare_images(x, y, nulls=x[spins],
                                nan_policy='omit', metric='pearsonr')
    return {'layer': layer, 'ctype': ctype, 'correlation': r, 'p_value': p}

def parallel_cross_layer_correlation(prop_mat, layer_CT, layers, ctypes, spins,
                                     use_clr=True, relative=True, mask=None,
                                     pseudocount=1e-8, fdr_alpha=0.05,
                                     max_workers=None, show_progress=True):
    """Spin-test correlation of layer composition vs. layer thickness.

    prop_mat : (n_region, n_layer, n_ctype)  within-layer proportions.
    layer_CT : (n_region, n_layer) DataFrame  absolute thickness.
    spins    : (n_spins, n_region) index array from spin_data(..., return_ind=True).
    use_clr  : CLR-transform composition per layer (recommended True).
    relative : row-normalise thickness across layers (recommended True).
    mask     : optional (layer x ctype) bool DataFrame; True == analyse.
    """
    n_region, n_layer, _ = prop_mat.shape
    assert layer_CT.shape == (n_region, n_layer), 'layer_CT shape mismatch'
    if mask is None:
        mask = pd.DataFrame(True, index=layers, columns=ctypes)
    mask = mask.astype(bool)

    # X: per-layer features. When use_clr, CLR is taken over the sub-composition of
    # PRESENT cell types per layer (mask) so structural zeros never enter the geometric
    # mean. Masked cells are also skipped in the task loop and dropped from the FDR family.
    X_layers = clr_features(prop_mat, layers, ctypes,
                            mask=mask if use_clr else None,
                            use_clr=use_clr, pseudocount=pseudocount)
    # Y: thickness (optionally relative)
    Y = layer_CT.div(layer_CT.sum(axis=1), axis=0) if relative else layer_CT
    Y = [Y.iloc[:, l].values for l in range(n_layer)]

    tasks = [(i, k, layer, ct, X_layers[i][ct].values, Y[i], spins)
             for i, layer in enumerate(layers)
             for k, ct in enumerate(ctypes) if mask.loc[layer, ct]]

    results = {}
    with ProcessPoolExecutor(max_workers=max_workers) as ex:
        fut = {ex.submit(_spin_single, t): n for n, t in enumerate(tasks)}
        it = as_completed(fut)
        if show_progress:
            from tqdm import tqdm
            it = tqdm(it, total=len(tasks), desc='spin')
        for f in it:
            results[fut[f]] = f.result()
    res = pd.DataFrame([results[n] for n in sorted(results)])

    ok = res['p_value'].notna()
    res.loc[ok, 'reject_H0'], res.loc[ok, 'p_adjusted'] = fdrcorrection(
        res.loc[ok, 'p_value'].values, alpha=fdr_alpha)
    return res

import math
from itertools import permutations
from scipy.stats import pearsonr

def _exact_mismatch(args):
    X_layers, Y_layers, i, k, ctype = args
    x, L = X_layers[i][ctype].values, len(Y_layers)
    r_obs, _ = pearsonr(x, Y_layers[i])
    perm_r = [pearsonr(x, Y_layers[perm[i]])[0]
              for perm in permutations(range(L)) if perm[i] != i]
    return {'layer': i, 'ctype': ctype, 'r_obs': r_obs,
            'p_value': np.mean(np.array(perm_r) >= r_obs),
            'n_perm_used': len(perm_r)}

def permutation_test_exact_mismatch(X_layers, Y_layers, ctypes, layers=None,
                                    mask=None, n_jobs=None, show_progress=True):
    """Exact enumeration mismatch permutation (fine for <= 8 layers).

    If mask (layers x ctypes bool, True=present) and layers are supplied, absent
    (structural-zero) cells are skipped and excluded from the FDR family.
    """
    L = len(X_layers)
    assert L == len(Y_layers), 'X_layers / Y_layers length mismatch'
    tasks = []
    for i in range(L):
        for k, ct in enumerate(ctypes):
            if ct not in X_layers[i].columns:                 # dropped by subcomposition CLR
                continue
            if mask is not None and layers is not None and not mask.loc[layers[i], ct]:
                continue
            tasks.append((X_layers, Y_layers, i, k, ct))
    n_draws = math.factorial(L) - math.factorial(L - 1)   # exclude perm[i]==i
    print(f'{len(tasks)} combinations x {n_draws} null draws each '
          f'({L-1} distinct mismatched layers)')
    with ProcessPoolExecutor(max_workers=n_jobs) as ex:
        futs = [ex.submit(_exact_mismatch, t) for t in tasks]
        it = as_completed(futs)
        if show_progress:
            from tqdm import tqdm
            it = tqdm(it, total=len(tasks), desc='exact-mismatch')
        res = [f.result() for f in it]
    df = pd.DataFrame(res)
    ok = df['p_value'].notna()
    df.loc[ok, 'reject_H0'], df.loc[ok, 'p_adjusted'] = fdrcorrection(
        df.loc[ok, 'p_value'].values, alpha=0.05)
    return df

def _ols_ci(x, y, n_pts=200, ci=0.95):
    n  = len(x);  xl = np.linspace(x.min(), x.max(), n_pts)
    sl, ic, *_ = linregress(x, y)
    yh = sl * x + ic;  s = np.sqrt(np.sum((y - yh) ** 2) / (n - 2))
    xm = x.mean();  sx = np.sum((x - xm) ** 2)
    se = s * np.sqrt(1 / n + (xl - xm) ** 2 / sx)
    tc = t_dist.ppf((1 + ci) / 2, df=n - 2)
    yl = sl * xl + ic
    return xl, yl, yl - tc * se, yl + tc * se

def helmert_basis(D):
    """
    Generate a Helmert contrast matrix (D x (D-1)) with orthonormal columns,
    each column sums to zero. This is a standard basis for ILR.
    
    Parameters:
        D (int): Number of components.
    
    Returns:
        np.ndarray: Shape (D, D-1), columns are orthonormal and sum to zero.
    """
    H = np.zeros((D, D-1))
    for k in range(D-1):
        # For column k, first k+1 entries are 1/sqrt((k+1)(k+2)), 
        # entry k+1 is -sqrt((k+1)/(k+2)), rest 0.
        # This is the standard Helmert matrix (column-wise).
        # We'll build row-wise for clarity.
        pass
    # Simpler: use the standard Helmert matrix from R or stats.
    # We'll construct using the recursive definition.
    # For D, we create a (D-1) x D matrix, then transpose.
    # Formula: for i=1..D-1, for j=1..i: value = 1/sqrt(i*(i+1)), 
    # for j=i+1: value = -i/sqrt(i*(i+1)), else 0.
    H = np.zeros((D-1, D))
    for i in range(1, D):
        H[i-1, :i] = 1 / np.sqrt(i * (i+1))
        H[i-1, i] = -i / np.sqrt(i * (i+1))
    return H.T  # shape (D, D-1)

def coda_transform(df, mode='ALR', ref=None, pseudocount=1e-8, basis=None):
    """
    ALR, CLR, or ILR transformation for compositional data.

    Parameters:
        df (pd.DataFrame): Original data, rows = samples, columns = features.
        mode (str): Transformation mode: 'ALR', 'CLR', or 'ILR'.
        ref (int or str, optional): Reference column for ALR only.
            - If None, select the column with the maximum median as reference.
            - If int or str, specify the column index or name.
        pseudocount (float): Small constant added to avoid log(0).
        basis (np.ndarray, optional): Custom basis matrix for ILR only.
            - Shape must be (D, D-1) with orthonormal columns, each summing to zero.
            - If None, the default Helmert basis is used.

    Returns:
        tuple: (df_transformed, ref_or_basis)
            - df_transformed: Transformed DataFrame.
            - ref_or_basis: Reference column name (for ALR), 'Geometric mean' (for CLR),
                            or the basis matrix used (for ILR).
    """

    cols = df.columns.tolist()
    if len(cols) < 2:
        raise ValueError("Need at least two features.")

    data = df[cols] + pseudocount
    D = len(cols)

    if mode.upper() == 'ALR':
        if ref is None:
            ref_col_name = data.median(axis=0).idxmax()
        elif isinstance(ref, int):
            ref_col_name = cols[ref]
        else:
            ref_col_name = ref
        if ref_col_name not in cols:
            raise ValueError(f"Reference column '{ref_col_name}' does not exist.")

        ref_vals = data[ref_col_name]
        transformed = pd.DataFrame(index=df.index)
        for col in cols:
            if col != ref_col_name:
                transformed[f'{col}'] = np.log(data[col] / ref_vals)
        return transformed, ref_col_name

    elif mode.upper() == 'CLR':
        geom_mean = gmean(data, axis=1)
        transformed = pd.DataFrame(index=df.index)
        for col in cols:
            transformed[f'{col}'] = np.log(data[col] / geom_mean)
        return transformed, 'Geometric mean'

    elif mode.upper() == 'ILR':
        # First compute CLR (centered log-ratio)
        geom_mean = gmean(data, axis=1)
        clr = np.log(data / geom_mean.values.reshape(-1, 1))  # shape (n, D)

        # Get basis matrix
        if basis is None:
            basis_matrix = helmert_basis(D)  # shape (D, D-1)
        else:
            basis_matrix = np.asarray(basis)
            if basis_matrix.shape != (D, D-1):
                raise ValueError(f"Basis matrix must be shape ({D}, {D-1}), got {basis_matrix.shape}")
            # Check orthonormality and column sums (optional, but we skip for speed)

        # ILR = CLR %*% basis_matrix  (n x (D-1))
        ilr_values = clr @ basis_matrix  # matrix multiplication

        # Create DataFrame with column names
        transformed = pd.DataFrame(ilr_values, index=df.index,
                                   columns=[f'ILR_{i+1}' for i in range(D-1)])
        return transformed, basis_matrix

    else:
        raise ValueError("mode must be 'ALR', 'CLR', or 'ILR'")

import numpy as np
import pandas as pd
from scipy.stats import gmean

def coda_transform(df, mode='ALR', ref=None, pseudocount=1e-8, basis=None):
    """
    ALR, CLR, or ILR transformation for compositional data.

    Parameters:
        df (pd.DataFrame): Original data, rows = samples, columns = features.
        mode (str): Transformation mode: 'ALR', 'CLR', or 'ILR'.
        ref (int or str, optional): Reference column for ALR only.
            - If None, select the column with the maximum median as reference.
            - If int or str, specify the column index or name.
        pseudocount (float): Small constant added to avoid log(0).
        basis (np.ndarray, optional): Custom basis matrix for ILR only.
            - Shape must be (D, D-1) with orthonormal columns, each summing to zero.
            - If None, the default sequential binary partition (SBP) basis is used,
              corresponding to the formula:
              ILR_i = sqrt(i/(i+1)) * ln( (prod_{j=1}^i x_j)^{1/i} / x_{i+1} ).

    Returns:
        tuple: (df_transformed, ref_or_basis)
            - df_transformed: Transformed DataFrame.
            - ref_or_basis: Reference column name (for ALR), 'Geometric mean' (for CLR),
                            or the basis matrix used (for ILR).
    """

    cols = df.columns.tolist()
    if len(cols) < 2:
        raise ValueError("Need at least two features.")

    data = df[cols] + pseudocount
    D = len(cols)

    if mode.upper() == 'ALR':
        if ref is None:
            ref_col_name = data.median(axis=0).idxmax()
        elif isinstance(ref, int):
            ref_col_name = cols[ref]
        else:
            ref_col_name = ref
        if ref_col_name not in cols:
            raise ValueError(f"Reference column '{ref_col_name}' does not exist.")

        ref_vals = data[ref_col_name]
        transformed = pd.DataFrame(index=df.index)
        for col in cols:
            if col != ref_col_name:
                transformed[f'{col}'] = np.log(data[col] / ref_vals)
        return transformed, ref_col_name

    elif mode.upper() == 'CLR':
        geom_mean = gmean(data, axis=1)
        transformed = pd.DataFrame(index=df.index)
        for col in cols:
            transformed[f'{col}'] = np.log(data[col] / geom_mean)
        return transformed, 'Geometric mean'

    elif mode.upper() == 'ILR':
        # If a custom basis is provided, use it; otherwise, use the default SBP basis.
        if basis is not None:
            basis_matrix = np.asarray(basis)
            if basis_matrix.shape != (D, D-1):
                raise ValueError(f"Basis matrix must be shape ({D}, {D-1}), got {basis_matrix.shape}")
        else:
            # Build the default basis matrix corresponding to the given formula.
            # Each column k (0-indexed) corresponds to i = k+1.
            basis_matrix = np.zeros((D, D-1))
            for i in range(1, D):          # i = 1 .. D-1
                k = i - 1                  # column index
                coef = np.sqrt(i / (i + 1))
                # For the first i components, weight = 1/i (geometric mean)
                basis_matrix[:i, k] = coef / i
                # For component i+1 (index i), weight = -coef
                basis_matrix[i, k] = -coef
                # The remaining components (i+2 .. D) get weight 0
            # The basis matrix columns are orthonormal and sum to zero.

        # Compute CLR (centered log-ratio)
        geom_mean = gmean(data, axis=1)
        clr = np.log(data / geom_mean.values.reshape(-1, 1))  # shape (n, D)

        # ILR = CLR %*% basis_matrix
        ilr_values = clr @ basis_matrix

        # Create DataFrame with column names
        transformed = pd.DataFrame(ilr_values, index=df.index,
                                   columns=[f'ILR_{i+1}' for i in range(D-1)])
        return transformed, basis_matrix

    else:
        raise ValueError("mode must be 'ALR', 'CLR', or 'ILR'")

def parallel_cross_layer_correlation(prop_mat, layer_CT, layers, ctypes, 
                                     spins, use_clr=True, pseudocount=1e-8,mask = None,
                                     relative=False, fdr_alpha=0.05, 
                                     max_workers=None, show_progress=True):
    """
    并行计算各层各特征的相关性。
    
    Parameters:
        prop_mat: (n_regions, n_layers, n_features)
        layer_CT: DataFrame (n_regions, n_layers) 或 2D array
        layers, ctypes: 名称列表
        spins: 置换索引矩阵 (n_spins, n_regions) —— 若 use_neuromaps=True，必须提供
        use_clr: 是否 CLR 转换
        relative: 是否对 layer_CT 进行行归一化
        use_neuromaps: True 使用 neuromaps.compare_images，False 使用自定义 permutation_correlation
        max_workers: 进程池最大工作数，默认 CPU 核数
        show_progress: 是否显示进度条（需要 tqdm）
    """
    n_features, n_regions, n_layers = prop_mat.shape
    if layer_CT.shape != (n_regions, n_layers):
        raise ValueError("layer_CT shape mismatch")
    if mask is not None:
        if not isinstance(mask, pd.DataFrame):
            raise TypeError("mask must be a pandas DataFrame")
        if not (mask.index == layers).all() or not (mask.columns == ctypes).all():
            raise ValueError("mask index/columns must match layers/ctypes")
        # 确保 mask 为布尔类型

import numpy as np
import pandas as pd
from scipy.stats import pearsonr
from concurrent.futures import ProcessPoolExecutor, as_completed
from tqdm import tqdm
import warnings
warnings.filterwarnings('ignore')

def single_combination_permutation(args):
    """
    单个组合 (layer_idx, ctype_idx) 的置换检验。
    返回该组合的结果字典。
    """
    X_layers, Y_layers, i, k, ctype, n_perm, seed = args
    np.random.seed(seed)
    
    # 提取 X 和原始 Y
    x = X_layers[i][ctype].values
    y_orig = Y_layers[i]
    # 剔除缺失值（同时移除 x 和 y_orig 的 NA）
    mask = ~(np.isnan(x) | np.isnan(y_orig))
    if np.sum(mask) < 3:
        return {'layer': i, 'ctype': ctype, 'r_obs': np.nan, 'p_value': np.nan, 'n_used': np.sum(mask)}
    x_clean = x[mask]
    y_clean = y_orig[mask]
    r_obs, _ = pearsonr(x_clean, y_clean)
    
    # 置换：打乱 Y 的层顺序，每次取第 i 层
    L = len(Y_layers)
    perm_r = np.zeros(n_perm)
    for t in range(n_perm):
        # 随机置换层索引
        perm_idx = np.random.permutation(L)
        # 取置换后的第 i 层
        y_perm_full = Y_layers[perm_idx[i]]
        # 对 x 和 y_perm 应用相同的 mask（剔除缺失值）
        # 但注意 mask 是基于原始 y_orig 的，置换后 y 可能不同，我们需要重新 mask
        # 更稳健：对每个置换单独 mask
        mask_perm = ~(np.isnan(x) | np.isnan(y_perm_full))
        if np.sum(mask_perm) < 3:
            perm_r[t] = np.nan
            continue
        x_perm = x[mask_perm]
        y_perm_clean = y_perm_full[mask_perm]
        r_perm, _ = pearsonr(x_perm, y_perm_clean)
        perm_r[t] = r_perm
    
    # 去除 NaN 的置换值
    perm_r = perm_r[~np.isnan(perm_r)]
    if len(perm_r) == 0:
        p_val = np.nan
    else:
        # 单侧：r_obs > r_perm 的比例
        p_val = np.mean(perm_r >= r_obs)
    
    return {
        'layer': i,
        'ctype': ctype,
        'r_obs': r_obs,
        'p_value': p_val,
        'mean_perm_r': np.mean(perm_r) if len(perm_r) > 0 else np.nan,
        'n_perm_used': len(perm_r)
    }

import numpy as np
import pandas as pd
from scipy.stats import pearsonr
from itertools import permutations
from concurrent.futures import ProcessPoolExecutor, as_completed
from tqdm import tqdm
import warnings
warnings.filterwarnings('ignore')

def single_exact_mismatch(args):
    """
    精确错配置换：枚举所有 L! 种层排列，取第 i 个位置作为错配层。
    排除自身匹配 (perm[i] == i)。
    """
    X_layers, Y_layers, i, k, ctype = args
    L = len(Y_layers)
    
    x = X_layers[i][ctype].values
    y_obs = Y_layers[i]
    
    # 1. 计算观测相关
    r_obs, _ = pearsonr(x, y_obs)
    
    # 2. 枚举所有层排列（L! 种）
    perm_r = []
    # 提前生成所有排列（若 L=6，仅720种，非常快）
    for perm in permutations(range(L)):
        j = perm[i]          # 打乱后第 i 个位置对应的是原始第几层
        if j == i:
            continue         # 剔除完全匹配（本层配本层），即 A^6_6 - 1
        y_perm = Y_layers[j]
        r_perm, _ = pearsonr(x, y_perm)
        perm_r.append(r_perm)
    
    # 3. 计算精确 p 值（单侧：错配 >= 本层的比例）
    # perm_r 的长度固定为 L! - 1
    p_val = np.mean(np.array(perm_r) >= r_obs)
    
    return {
        'layer': i,
        'ctype': k,
        'r_obs': r_obs,
        'p_value': p_val,
        'n_perm_used': len(perm_r)  # 固定为 719
    }


def permutation_test_exact_mismatch(X_layers, Y_layers, ctypes, 
                                    n_jobs=None, show_progress=True):
    """
    精确枚举错配置换检验（适合层数 <= 8）。
    """
    L = len(X_layers)
    if L != len(Y_layers):
        raise ValueError("X_layers and Y_layers length mismatch")
    if L > 8:
        print(f"Warning: L={L}, total permutations {np.math.factorial(L)} might be large. Consider random sampling.")
    
    # 生成所有任务
    tasks = []
    for i in range(L):
        for k, ctype in enumerate(ctypes):
            tasks.append((X_layers, Y_layers, i, k, ctype))
    
    print(f"Total combinations: {len(tasks)}")
    print(f"Each combination enumerates {np.math.factorial(L)} permutations, using {np.math.factorial(L)-1} mismatches.")
    
    results = []
    with ProcessPoolExecutor(max_workers=n_jobs) as executor:
        if show_progress:
            futures = [executor.submit(single_exact_mismatch, task) for task in tasks]
            for future in tqdm(as_completed(futures), total=len(tasks), desc="Exact Permutations"):
                try:
                    results.append(future.result())
                except Exception as e:
                    print(f"Task failed: {e}")
                    results.append(None)
        else:
            results = list(executor.map(single_exact_mismatch, tasks))
    
    results = [r for r in results if r is not None]
    df_results = pd.DataFrame(results)
    
    # FDR 校正
    from statsmodels.stats.multitest import fdrcorrection
    valid = ~np.isnan(df_results['p_value'])
    if valid.any():
        reject, p_adj = fdrcorrection(df_results.loc[valid, 'p_value'].values, alpha=0.05)
        df_results.loc[valid, 'p_adjusted'] = p_adj
        df_results.loc[valid, 'reject_H0'] = reject
    else:
        df_results['p_adjusted'] = np.nan
        df_results['reject_H0'] = False
    
    return df_results

                rgba = cmap(norm(val))
                text_color = get_text_color(rgba)
                ax.text(j, i, symbol, ha='center', va='center',
                        color=text_color, fontsize=8, fontweight='bold', zorder=20)

    # ----- 6. Axes styling -----
    ax.set_xticks(np.arange(n_cols))
    ax.set_xticklabels(final_cols, rotation=90, ha='center', va='top', fontsize=8)
    ax.yaxis.tick_right()
    ax.yaxis.set_label_position('right')
    ax.set_yticks(np.arange(n_rows))
    ax.set_yticklabels(layer_names_roman, fontsize=8, va='center')
    ax.tick_params(axis='y', length=0)

    for spine in ax.spines.values():
        spine.set_visible(False)

    # ----- 7. Top annotation bar -----
    top_ax = ax.inset_axes([0, 1.02, 1, 0.1], transform=ax.transAxes)
    top_ax.set_xlim(0, n_cols)
    top_ax.set_ylim(0, 1)
    top_ax.axis('off')
    for cls, (start, end) in cls_positions.items():
        color = cls_colors.get(cls, '#888888')
        top_ax.add_patch(mpatches.Rectangle((start, 0), end-start, 1, facecolor=color, edgecolor='none'))
        if end - start > 2:
            top_ax.text((start+end)/2, 0.5, cls, ha='center', va='center',
                        fontsize=9, color='white', fontweight='bold')

    # ----- 8. Colorbar -----
    cbar = fig.colorbar(im, ax=ax, orientation='vertical', fraction=0.1, pad=0.08,
                        shrink=0.8, aspect=10)
    cbar.set_label(cbar_label, fontsize=9, labelpad=8)
    cbar.ax.tick_params(labelsize=7, length=3, width=0.6)
    ticks = np.arange(vmin, vmax+0.01, (vmax-vmin)/4)
    cbar.set_ticks(ticks)
    cbar.ax.set_yticklabels([f'{t:.2f}' for t in ticks])

    # ----- 9. Title -----
    ax.set_title(title, fontsize=11, pad=18, weight='normal')

    # ----- 10. Save or show -----
    plt.tight_layout()
    if output_file:
        plt.savefig(output_file, dpi=dpi, bbox_inches='tight')
    plt.show()
    return fig, ax
